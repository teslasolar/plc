📚 Quick Start

const plc = new PLC('Mitsu', 1);
plc.init();
plc.addRung(
  'this.tags["X0"] && this.tags["M0"]',
  function() {
    this.tags["Y0"] = true;
  }
);
plc.setState('EXECUTE');
