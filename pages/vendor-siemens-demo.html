📚 Quick Start

const plc = new PLC('Siemens', 1);
plc.init();
plc.addRung(
  'this.tags["I0.0"] && this.tags["M0.0"]',
  function() {
    this.tags["Q0.0"] = true;
  }
);
plc.setState('EXECUTE');

📖 STL Instructions

A - AND logic
O - OR logic
AN - AND NOT
L - Load
T - Transfer
= - Assignment
